     1→/**
     2→ * Check engine: converts a page fetch result + inspection into structured,
     3→ * evidence-backed findings.
     4→ *
     5→ * Severity model (deterministic — see docs/ARCHITECTURE.md):
     6→ *   CRITICAL — confirmed 404/410/5xx, DNS/connect/TLS failure, timeout,
     7→ *              redirect loop / limit, strong campaign-parameter loss,
     8→ *              platform-relevant tracking absence with NO tag manager.
     9→ *   WARNING  — 403/429 scanner access, unusual 4xx, hostname change to a
    10→ *              different domain, tracking needs verification, sold-out /
    11→ *              soft-404 / maintenance heuristics, slow response, noindex,
    12→ *              missing title, thin content, unmet expectations.
    13→ *   INFO     — detected tags, preserved attribution across redirects,
    14→ *              CTA/form absence when not expected.
    15→ *
    16→ * Claim discipline: we write "not detected", never "missing", unless a
    17→ * runtime check confirmed it. Associated spend is described as spend
    18→ * associated with affected destinations — never as proven revenue loss.
    19→ *
    20→ * Every finding carries a stable `key` so its identity is comparable across
    21→ * scans (incident history: first seen / last seen / resolved).
    22→ */
    23→
    24→import { PRODUCT } from "@/config/product";
    25→import { inspectContent } from "./content";
    26→import { detectTrackers, platformRelevantTrackers, trackerLabel } from "./tracking";
    27→import { diffAttribution, sameSite } from "@/lib/urls/normalize";
    28→import type {
    29→  Confidence,
    30→  EvidenceItem,
    31→  FindingDraft,
    32→  PageFetchResult,
    33→  ScanContext,
    34→  Severity,
    35→  TrackerDetection,
    36→  TrackerKey,
    37→} from "./types";

    39→
    40→function fmtSpend(minor: number, currency: string): string {
    41→  return new Intl.NumberFormat("en-GB", { style: "currency", currency }).format(minor / 100);
    42→}
    43→
    44→/**
    45→ * Run all checks for one destination and return findings.
    46→ * `fetchResult` is injected: a live fetch, a fixture, or a test stub.
    47→ */
    48→export function runChecks(fetchResult: PageFetchResult, ctx: ScanContext): {
    49→  findings: FindingDraft[];
    50→  trackers: TrackerDetection[];
    51→} {
    52→  const findings: FindingDraft[] = [];
    53→
    54→  if (fetchResult.kind === "error") {
    55→    findings.push(...networkErrorFindings(fetchResult, ctx));
    56→    return { findings, trackers: [] };
    57→  }
    58→
    59→  // Page-level checks (tracking, content, expectations) only apply to an
    60→  // inspectable HTML page: a 2xx HTML response with body content. Error
    61→  // pages must not produce "tracking not detected" findings.
    62→  const contentType = fetchResult.contentType ?? "";
    63→  const inspectable =
    64→    fetchResult.httpStatus >= 200 &&
    65→    fetchResult.httpStatus < 300 &&
    66→    fetchResult.html.length > 0 &&
    67→    (/text\/html|application\/xhtml/i.test(contentType) || contentType === "");
    68→
    69→  const trackers = inspectable ? detectTrackers(fetchResult.html) : [];
    70→  const content = inspectable ? inspectContent(fetchResult.html) : null;
    71→
    72→  findings.push(...networkStatusFindings(fetchResult, ctx));
    73→  findings.push(...redirectFindings(fetchResult, ctx));
    74→  findings.push(...attributionFindings(fetchResult, ctx));
    75→  if (inspectable && content) {
    76→    findings.push(...trackingFindings(ctx, trackers));
    77→    findings.push(...contentFindings(fetchResult, ctx, content));
    78→    findings.push(...expectationFindings(fetchResult, ctx, content, trackers));
    79→  }
    80→  findings.push(...performanceFindings(fetchResult, ctx));
    81→
    82→  return { findings, trackers };
    83→}

    89→function networkErrorFindings(fetch: Extract<PageFetchResult, { kind: "error" }>, ctx: ScanContext): FindingDraft[] {
    90→  const spendText = fmtSpend(ctx.associatedSpendMinor, ctx.currency);
    91→  const evidence: EvidenceItem[] = [
    92→    { type: "FINAL_URL", label: "Requested URL", value: fetch.requestedUrl },
    93→  ];
    94→
    95→  const critical = (key: string, title: string, message: string, explanation: string, recommendation: string, extra: EvidenceItem[] = []): FindingDraft => ({
    96→    key,
    97→    checkId: CHECK_IDS.network,
    98→    severity: "critical",
    99→    confidence: "confirmed",
   100→    title,
   101→    summary: message,
   102→    explanation,
   103→    recommendation,
   104→    evidence: [...evidence, ...extra],
   105→    metadata: { errorCode: fetch.errorCode },
   106→  });
   107→
   108→  switch (fetch.errorCode) {
   109→    case "DNS_ERROR":
   110→      return [
   111→        critical(
   112→          "error_dns",
   113→          "Destination hostname could not be resolved",
   114→          `The scanner could not resolve this hostname (DNS). ${spendText} of campaign spend points here.`,
   115→          "The hostname in the imported URL did not resolve to any address. The destination may have been retired, or the campaign rows contain a typo in the domain.",
   116→          "Check the hostname in the imported campaign rows. If the domain was retired, replace the campaign destination before launch.",
   117→          [{ type: "DNS_ERROR", label: "DNS result", value: "No address found", meta: { detail: fetch.detail ?? undefined } }]
   118→        ),
   119→      ];
   120→    case "TLS_ERROR":
   121→      return [
   122→        critical(
   123→          "error_tls",
   124→          "TLS connection failed",
   125→          `The scanner could not establish a TLS connection to this destination. ${spendText} of campaign spend points here.`,
   126→          "The request failed during the TLS handshake. Common causes are an expired, self-signed, or domain-mismatched certificate.",
   127→          "Inspect the certificate on this destination (for example with your browser's security panel) and renew or correct it before launch.",
   128→          [{ type: "TLS_ERROR", label: "TLS failure", value: fetch.detail ?? fetch.message }]
   129→        ),
   130→      ];
   131→    case "CONNECTION_FAILURE":
   132→      return [
   133→        critical(
   134→          "error_connection",
   135→          "The scanner could not connect to this URL",
   136→          `The connection could not be established. ${spendText} of campaign spend points here.`,
   137→          "The server refused the connection or the network reported the host as unreachable. The destination is not serving traffic to the scanner.",
   138→          "Verify the destination in a normal browser. If it fails there too, restore the service or replace the campaign destination before launch.",
   139→          [{ type: "CONNECTION_ERROR", label: "Connection failure", value: fetch.message }]
   140→        ),
   141→      ];
   142→    case "TIMEOUT":
   143→      return [
   144→        critical(
   145→          "error_timeout",
   146→          "The scanner stopped this request after the timeout",
   147→          `The request did not complete within ${Math.round(PRODUCT.scanner.timeoutMs / 1000)} seconds. ${spendText} of campaign spend points here.`,
   148→          "The server accepted the connection but did not return a response in time. This is measured from the scanner; a normal browser may behave differently if the slowness is intermittent.",
   149→          "Verify response time from a normal browser and your monitoring. A destination that regularly exceeds this time will lose paid clicks.",
   150→          [
   151→            {
   152→              type: "TIMEOUT",
   153→              label: "Timeout",
   154→              value: `${PRODUCT.scanner.timeoutMs} ms`,
   155→              meta: { elapsedMs: fetch.responseTimeMs },
   156→            },
   157→          ]
   158→        ),
   159→      ];
   160→    case "BLOCKED_URL":
   161→      return [
   162→        {
   163→          key: "error_blocked",
   164→          checkId: CHECK_IDS.network,
   165→          severity: "info",
   166→          confidence: "confirmed",
   167→          title: "Destination blocked by scanner safety rules",
   168→          summary: "The scanner refused to request this destination for safety. It was not inspected.",
   169→          explanation:
   170→            "The destination resolves to an address the scanner treats as internal or unsafe (for example loopback, private, or link-local ranges). LandingSentinel never requests such destinations.",
   171→          recommendation:
   172→            "If you expected this destination to be a public page, check the imported URL for typos. Internal addresses cannot be inspected by the scanner.",
   173→          evidence: [
   174→            { type: "URL_BLOCKED", label: "Blocked destination", value: fetch.requestedUrl, meta: { reason: fetch.detail ?? fetch.message } },
   175→          ],
   176→          metadata: { errorCode: fetch.errorCode },
   177→        },
   178→      ];
   179→    case "REDIRECT_LOOP":
   180→      return [
   181→        critical(
   182→          "error_redirect_loop",
   183→          "This destination redirects in a loop",
   184→          `The scanner stopped because the redirect chain returned to an earlier URL. ${spendText} of campaign spend points here.`,
   185→          "Each redirect in the chain was followed manually until a URL repeated. Browsers will behave the same way and show an error page.",
   186→          "Fix the redirect rules so the chain terminates on a real page before launch.",
   187→          [
   188→            { type: "REDIRECT_LOOP", label: "Loop detected after", value: `${fetch.redirects.length} redirects` },
   189→            ...fetch.redirects.map((h) => hopEvidence(h)),
   190→          ]
   191→        ),
   192→      ];
   193→    case "TOO_MANY_REDIRECTS":
   194→      return [
   195→        critical(
   196→          "error_too_many_redirects",
   197→          "Redirect limit exceeded",
   198→          `The scanner stopped after ${PRODUCT.scanner.maxRedirects} redirects. ${spendText} of campaign spend points here.`,
   199→          "The destination kept redirecting beyond the scanner's redirect budget. Long chains add delay and can strip parameters before the visitor reaches a page.",
   200→          "Shorten the redirect chain so visitors land directly on the final page.",
   201→          fetch.redirects.map((h) => hopEvidence(h)),
   202→        ),
   203→      ];
   204→    default:
   205→      return [
   206→        critical(
   207→          "error_scan",
   208→          "The scanner could not complete this request",
   209→          `Reason: ${fetch.message}`,
   210→          "The request failed before a response could be inspected. The destination was not verified as working.",
   211→          "Retry the scan. If this repeats, verify the destination in a normal browser.",
   212→          [{ type: "CONNECTION_ERROR", label: "Request failure", value: fetch.message, meta: { detail: fetch.detail ?? undefined } }]
   213→        ),
   214→      ];
   215→  }
   216→}
   217→
   218→function hopEvidence(h: { sequence: number; fromUrl: string; toUrl: string; statusCode: number; durationMs: number | null }): EvidenceItem {
   219→  return {
   220→    type: "REDIRECT_HOP",
   221→    label: `Redirect ${h.sequence + 1}`,
   222→    value: `${h.fromUrl} → ${h.toUrl}`,
   223→    meta: { statusCode: h.statusCode, durationMs: h.durationMs },
   224→  };
   225→}

   231→function networkStatusFindings(fetch: Extract<PageFetchResult, { kind: "ok" }>, ctx: ScanContext): FindingDraft[] {
   232→  const spendText = fmtSpend(ctx.associatedSpendMinor, ctx.currency);
   233→  const status = fetch.httpStatus;
   234→  const baseEvidence: EvidenceItem[] = [
   235→    { type: "HTTP_STATUS", label: "HTTP status", value: String(status) },
   236→    { type: "FINAL_URL", label: "Final URL", value: fetch.finalUrl },
   237→    { type: "RESPONSE_TIME", label: "Response time", value: `${fetch.responseTimeMs} ms` },
   238→  ];
   239→
   240→  if (status === 404 || status === 410) {
   241→    return [
   242→      {
   243→        key: `status_${status}`,
   244→        checkId: CHECK_IDS.network,
   245→        severity: "critical",
   246→        confidence: "confirmed",
   247→        title: `Destination returned HTTP ${status}`,
   248→        summary: `${spendText} of campaign spend points at a destination the server reports as ${status === 404 ? "not found" : "gone"}.`,
   249→        explanation:
   250→          status === 404
   251→            ? "The server responded that this page does not exist. Visitors from these campaigns receive an error page."
   252→            : "The server responded that this page has been permanently removed. Visitors from these campaigns receive an error page.",
   253→        recommendation: "Replace the campaign destination or restore the landing page before launch.",
   254→        evidence: baseEvidence,
   255→      },
   256→    ];
   257→  }
   258→
   259→  if (status >= 500) {
   260→    return [
   261→      {
   262→        key: `status_${status}`,
   263→        checkId: CHECK_IDS.network,
   264→        severity: "critical",
   265→        confidence: "confirmed",
   266→        title: `Destination returned HTTP ${status}`,
   267→        summary: `The server returned a ${status} server error. ${spendText} of campaign spend points here.`,
   268→        explanation: "The origin server failed to produce the page. Visitors from these campaigns receive an error page while the error persists.",
   269→        recommendation: "Check the origin server or hosting for this destination before launch, then re-scan to confirm it returns a valid page.",
   270→        evidence: baseEvidence,
   271→      },
   272→    ];
   273→  }
   274→
   275→  if (status === 403 || status === 429) {
   276→    return [
   277→      {
   278→        key: `status_${status}`,
   279→        checkId: CHECK_IDS.network,
   280→        severity: "warning",
   281→        confidence: "confirmed",
   282→        title: `The scanner received HTTP ${status}`,
   283→        summary: `${spendText} of campaign spend points here. The page was not fully inspected.`,
   284→        explanation:
   285→          "The site can block automated requests. This result does not show that normal visitors cannot access the page. Verify the URL in a normal browser before treating it as a failure.",
   286→        recommendation:
   287→          "Open the destination in a normal browser. If it loads, the finding reflects scanner access only — you can allow the scanner user agent on your site.",
   288→        evidence: baseEvidence,
   289→      },
   290→    ];
   291→  }
   292→
   293→  if (status >= 400) {
   294→    return [
   295→      {
   296→        key: `status_${status}`,
   297→        checkId: CHECK_IDS.network,
   298→        severity: "warning",
   299→        confidence: "confirmed",
   300→        title: `The scanner received HTTP ${status}`,
   301→        summary: `${spendText} of campaign spend points here. An unexpected client error was returned.`,
   302→        explanation: "The server returned an error status that is unusual for a campaign destination. The page content could not be inspected.",
   303→        recommendation: "Verify the destination in a normal browser before launch.",
   304→        evidence: baseEvidence,
   305→      },
   306→    ];
   307→  }
   308→
   309→  const contentType = fetch.contentType ?? "";
   310→  const isHtml = contentType === "" || /text\/html|application\/xhtml/i.test(contentType);
   311→  if (!isHtml) {
   312→    return [
   313→      {
   314→        key: "non_html_response",
   315→        checkId: CHECK_IDS.network,
   316→        severity: "warning",
   317→        confidence: "confirmed",
   318→        title: "Destination did not return HTML",
   319→        summary: `The response content type was "${contentType || "unknown"}". ${spendText} of campaign spend points here.`,
   320→        explanation:
   321→          "The scanner inspects HTML landing pages. This destination returned another content type (for example a PDF or JSON feed), so page-level checks were skipped.",
   322→        recommendation: "Confirm whether this destination is intended. Campaign landing pages are normally HTML documents.",
   323→        evidence: [
   324→          ...baseEvidence,
   325→          { type: "CONTENT_TYPE", label: "Content type", value: contentType || "unknown" },
   326→        ],
   327→      },
   328→    ];
   329→  }
   330→  return [];
   331→}

   336→
   337→function redirectFindings(fetch: Extract<PageFetchResult, { kind: "ok" }>, ctx: ScanContext): FindingDraft[] {
   338→  const findings: FindingDraft[] = [];
   339→  if (fetch.redirects.length === 0) return findings;
   340→
   341→  const hops = fetch.redirects.map(hopEvidence);
   342→  const spendText = fmtSpend(ctx.associatedSpendMinor, ctx.currency);
   343→
   344→  try {
   345→    const from = new URL(fetch.requestedUrl);
   346→    const to = new URL(fetch.finalUrl);
   347→    const hostChanged = from.hostname !== to.hostname;
   348→    if (hostChanged) {
   349→      const differentDomain = !sameSite(from.hostname, to.hostname);
   350→      findings.push({
   351→        key: differentDomain ? "hostname_change_cross_domain" : "hostname_change_same_site",
   352→        checkId: CHECK_IDS.redirects,
   353→        severity: differentDomain ? "warning" : "info",
   354→        confidence: "confirmed",
   355→        title: differentDomain ? "Final destination is on a different domain" : "Hostname changed after redirect",
   356→        summary: `The imported destination redirects to ${to.hostname}. ${spendText} of campaign spend points at the imported URL.`,
   357→        explanation: differentDomain
   358→          ? "Visitors still reach a page, but on a domain other than the one in the campaign rows. This can affect measurement set up for the original domain, and any server-side rules tied to it."
   359→          : "The page is served from a different hostname of the same site (for example www and non-www forms). This is usually intentional.",
   360→        recommendation: differentDomain
   361→          ? "Confirm the cross-domain redirect is intentional and that measurement follows the visitor across domains."
   362→          : undefined,
   363→        evidence: [...hops, { type: "FINAL_URL", label: "Final URL", value: fetch.finalUrl }],
   364→      });
   365→    }
   366→  } catch {
   367→    /* malformed URLs are handled elsewhere */
   368→  }
   369→
   370→  return findings;
   371→}

   373→/* -------------------------------------------------------------- */
   374→/* Attribution / parameter survival                               */
   375→/* -------------------------------------------------------------- */
   376→
   377→const PRIMARY_UTM = ["utm_campaign", "utm_source", "utm_medium"];
   378→const CLICK_IDS = ["gclid", "gbraid", "wbraid", "fbclid", "msclkid", "ttclid", "li_fat_id"];
   379→
   380→function attributionFindings(fetch: Extract<PageFetchResult, { kind: "ok" }>, ctx: ScanContext): FindingDraft[] {
   381→  const findings: FindingDraft[] = [];
   382→  if (fetch.redirects.length === 0) return findings;
   383→
   384→  let diffs;
   385→  try {
   386→    diffs = diffAttribution(fetch.requestedUrl, fetch.finalUrl);
   387→  } catch {
   388→    return findings;
   389→  }
   390→
   391→  const removedPrimary = diffs.filter((d) => d.status === "removed" && PRIMARY_UTM.includes(d.param));
   392→  const removedSecondary = diffs.filter((d) => d.status === "removed" && (d.param === "utm_content" || d.param === "utm_term"));
   393→  const removedClickIds = diffs.filter((d) => d.status === "removed" && CLICK_IDS.includes(d.param));
   394→  const changed = diffs.filter((d) => d.status === "changed");
   395→  const preserved = diffs.filter((d) => d.status === "preserved");
   396→
   397→  const spendText = fmtSpend(ctx.associatedSpendMinor, ctx.currency);
   398→  const hopList = fetch.redirects.map(hopEvidence);
   399→
   400→  if (removedPrimary.length > 0) {
   401→    const param = removedPrimary[0];
   402→    findings.push({
   403→      key: "param_removed_primary",
   404→      checkId: CHECK_IDS.attribution,
   405→      severity: "critical",
   406→      confidence: "confirmed",
   407→      title: "Campaign parameter removed after redirect",
   408→      summary: `The imported destination contained ${param.param}=${param.originalValue}. The final URL does not contain this parameter. ${spendText} of campaign spend points here.`,
   409→      explanation:
   410→        "The redirect chain drops a primary campaign parameter. Analytics that read UTM parameters from the landing URL will not attribute these visits to the imported campaign values.",
   411→      recommendation:
   412→        "Verify whether attribution is preserved through another mechanism (for example server-side tagging). If it is not, update the redirect or the campaign destination before launch.",
   413→      evidence: [
   414→        {
   415→          type: "PARAMETER_REMOVED",
   416→          label: `Removed parameter: ${param.param}`,
   417→          value: `${param.param}=${param.originalValue ?? ""}`,
   418→          meta: { finalUrl: fetch.finalUrl, redirects: fetch.redirects.length },
   419→        },
   420→        ...hopList,
   421→        { type: "FINAL_URL", label: "Final URL", value: fetch.finalUrl },
   422→      ],
   423→      metadata: { removedParams: removedPrimary.map((d) => d.param) },
   424→    });
   425→  }
   426→
   427→  if (removedSecondary.length > 0) {
   428→    findings.push({
   429→      key: "param_removed_secondary",
   430→      checkId: CHECK_IDS.attribution,
   431→      severity: "warning",
   432→      confidence: "confirmed",
   433→      title: "Secondary campaign parameter removed after redirect",
   434→      summary: `${removedSecondary.map((d) => d.param).join(", ")} ${removedSecondary.length === 1 ? "was" : "were"} present in the imported URL and absent from the final URL.`,
   435→      explanation:
   436→        "These parameters support content- and term-level reporting. Their removal limits ad-level or keyword-level analysis in analytics that reads them from the URL.",
   437→      recommendation: "Verify whether your analytics setup preserves these values by another method.",
   438→      evidence: removedSecondary.map((d) => ({
   439→        type: "PARAMETER_REMOVED" as const,
   440→        label: `Removed parameter: ${d.param}`,
   441→        value: `${d.param}=${d.originalValue ?? ""}`,
   442→        meta: { finalUrl: fetch.finalUrl },
   443→      })),
   444→    });
   445→  }
   446→
   447→  if (removedClickIds.length > 0) {
   448→    findings.push({
   449→      key: "param_removed_click_id",
   450→      checkId: CHECK_IDS.attribution,
   451→      severity: "warning",
   452→      confidence: "needs_verification",
   453→      title: "Click ID removed after redirect",
   454→      summary: `${removedClickIds.map((d) => d.param).join(", ")} ${removedClickIds.length === 1 ? "was" : "were"} in the imported URL and absent from the final URL.`,
   455→      explanation:
   456→        "Click IDs normally travel in the landing URL and are captured for platform-side conversion measurement. A redirect that strips them can still be correct if the platform redirects them onward itself or measurement is server-side.",
   457→      recommendation:
   458→        "Verify with your tag debugger or the platform's conversion diagnostics whether click IDs are captured for this destination.",
   459→      evidence: removedClickIds.map((d) => ({
   460→        type: "PARAMETER_REMOVED" as const,
   461→        label: `Removed parameter: ${d.param}`,
   462→        value: `${d.param}=${d.originalValue ?? ""}`,
   463→        meta: { finalUrl: fetch.finalUrl },
   464→      })),
   465→    });
   466→  }
   467→
   468→  for (const change of changed) {
   469→    findings.push({
   470→      key: `param_changed_${change.param}`,
   471→      checkId: CHECK_IDS.attribution,
   472→      severity: "warning",
   473→      confidence: "confirmed",
   474→      title: "Campaign parameter changed after redirect",
   475→      summary: `${change.param} was "${change.originalValue}" in the imported URL and "${change.finalValue}" in the final URL.`,
   476→      explanation: "A redirect rewrote the parameter value. Analytics will attribute these visits to the new value, not the value in your campaign rows.",
   477→      recommendation: "Check the redirect rules for this destination if the original value is the intended one.",
   478→      evidence: [
   479→        {
   480→          type: "PARAMETER_CHANGED",
   481→          label: `Changed parameter: ${change.param}`,
   482→          value: `${change.originalValue} → ${change.finalValue}`,
   483→        },
   484→        ...hopList,
   485→      ],
   486→    });
   487→  }
   488→
   489→  if (findings.length === 0 && preserved.length > 0) {
   490→    findings.push({
   491→      key: "attribution_preserved",
   492→      checkId: CHECK_IDS.attribution,
   493→      severity: "info",
   494→      confidence: "confirmed",
   495→      title: "Redirect occurred, attribution parameters preserved",
   496→      summary: `The destination redirected ${fetch.redirects.length} ${fetch.redirects.length === 1 ? "time" : "times"} and all ${preserved.length} campaign parameters survived to the final URL.`,
   497→      explanation:
   498→        "The final URL still contains the imported campaign parameters, so URL-based analytics attribution is not interrupted by the redirect.",
   499→      evidence: [
   500→        ...preserved.map((p) => ({
   501→          type: "PARAMETER_PRESERVED" as const,
   502→          label: `Preserved: ${p.param}`,
   503→          value: `${p.param}=${p.originalValue ?? ""}`,
   504→        })),
   505→        ...hopList,
   506→      ],
   507→    });
   508→  }
   509→
   510→  return findings;
   511→}

   516→
   517→function trackingEvidenceLine(trackers: TrackerDetection[], key: TrackerKey): EvidenceItem {
   518→  const t = trackers.find((tr) => tr.key === key);
   519→  return {
   520→    type: t?.detected ? "TRACKER_SIGNATURE" : "TRACKER_ABSENT",
   521→    label: trackerLabel(key),
   522→    value: t?.detected ? `Detected — signature "${t.signatureId}"` : "Not detected in retrieved HTML",
   523→    snippet: t?.excerpt ?? undefined,
   524→  };
   525→}
   526→
   527→function trackingFindings(ctx: ScanContext, trackers: TrackerDetection[]): FindingDraft[] {
   528→  const findings: FindingDraft[] = [];
   529→  const gtmPresent = trackers.find((t) => t.key === "gtm")?.detected ?? false;
   530→  const spendText = fmtSpend(ctx.associatedSpendMinor, ctx.currency);
   531→
   532→  const platforms = [...new Set(ctx.platforms)];
   533→  for (const platform of platforms) {
   534→    const relevant = platformRelevantTrackers(platform);
   535→    if (relevant.length === 0) continue;
   536→
   537→    const anyDetected = relevant.some((key) => trackers.find((t) => t.key === key)?.detected);
   538→    if (anyDetected) continue;
   539→
   540→    const platformName: Record<string, string> = {
   541→      meta: "Meta",
   542→      google: "Google Ads",
   543→      tiktok: "TikTok",
   544→      linkedin: "LinkedIn",
   545→    };
   546→    const platformLabel = platformName[platform] ?? platform;
   547→    const relevantLabels = relevant.map((k) => trackerLabel(k)).join(" / ");
   548→
   549→    const baseEvidence: EvidenceItem[] = [
   550→      { type: "TRACKER_ABSENT", label: `Imported platform: ${platformLabel}`, value: `${platformLabel} campaign spend is associated with this destination` },
   551→      ...relevant.map((k) => trackingEvidenceLine(trackers, k)),
   552→      trackingEvidenceLine(trackers, "gtm"),
   553→    ];
   554→
   555→    if (!gtmPresent) {
   556→      findings.push({
   557→        key: `tracking_absent_${platform}`,
   558→        checkId: CHECK_IDS.tracking,
   559→        severity: "critical",
   560→        confidence: "high",
   561→        title: `${platformLabel} tracking was not detected`,
   562→        summary: `${spendText} of ${platformLabel} campaign spend points at this destination, and no ${platformLabel} tracking signature or tag manager was detected in the retrieved HTML.`,
   563→        explanation: `Imported platform: ${platformLabel}. ${relevantLabels}: not detected. Google Tag Manager: not detected. Static HTML scanning cannot prove what happens after JavaScript runs, but with no direct signature and no tag manager present, nothing in the retrieved page would load a ${platformLabel} tag.`,
   564→        recommendation: `Open the rendered page with a tag debugger (for example ${platformLabel === "Meta" ? "Meta Pixel Helper" : "the platform's tag diagnostics"}) and confirm whether tracking loads at runtime. If it does not, install tracking before spending against this destination.`,
   565→        evidence: baseEvidence,
   566→        metadata: { platform, relevantTrackers: relevant },
   567→      });
   568→    } else {
   569→      findings.push({
   570→        key: `tracking_absent_${platform}`,
   571→        checkId: CHECK_IDS.tracking,
   572→        severity: "warning",
   573→        confidence: "needs_verification",
   574→        title: `${relevantLabels} was not detected directly`,
   575→        summary: `${spendText} of ${platformLabel} campaign spend points here. Google Tag Manager is present; a tag can load at runtime.`,
   576→        explanation: `Google Tag Manager is detected in the retrieved HTML. A ${relevantLabels} can load through GTM after the page runs. This static scan cannot confirm that behavior.`,
   577→        recommendation: `Open the rendered page with a tag debugger and verify the ${relevantLabels} fires before you treat this as a confirmed tracking failure.`,
   578→        evidence: baseEvidence,
   579→        metadata: { platform, relevantTrackers: relevant, gtmPresent: true },
   580→      });
   581→    }
   582→  }
   583→
   584→  const detected = trackers.filter((t) => t.detected);
   585→  if (detected.length > 0) {
   586→    findings.push({
   587→      key: "tracking_detected",
   588→      checkId: CHECK_IDS.tracking,
   589→      severity: "info",
   590→      confidence: "confirmed",
   591→      title: "Tracking signatures detected",
   592→      summary: `${detected.length} tracking ${detected.length === 1 ? "signature was" : "signatures were"} detected in the retrieved HTML.`,
   593→      explanation:
   594→        "Static detection proves the signature exists in the served HTML. It does not confirm that tags fire at runtime or that they are configured correctly.",
   595→      evidence: detected.map((t) => ({
   596→        type: "TRACKER_SIGNATURE",
   597→        label: trackerLabel(t.key),
   598→        value: `Detected — signature "${t.signatureId}"`,
   599→        snippet: t.excerpt ?? undefined,
   600→      })),
   601→    });
   602→  }
   603→
   604→  return findings;
   605→}

   610→
   611→function contentFindings(
   612→  fetch: Extract<PageFetchResult, { kind: "ok" }>,
   613→  ctx: ScanContext,
   614→  content: ReturnType<typeof inspectContent>
   615→): FindingDraft[] {
   616→  const findings: FindingDraft[] = [];
   617→  const spendText = fmtSpend(ctx.associatedSpendMinor, ctx.currency);
   618→
   619→  if (!content.title) {
   620→    findings.push({
   621→      key: "title_missing",
   622→      checkId: CHECK_IDS.content,
   623→      severity: "warning",
   624→      confidence: "confirmed",
   625→      title: "HTML title is missing or empty",
   626→      summary: `The retrieved page has no <title> content. ${spendText} of campaign spend points here.`,
   627→      explanation:
   628→        "The title element is empty or absent. Titles are used by browsers, search results, and some link previews. An empty title is a content defect on a paid landing page.",
   629→      recommendation: "Add a descriptive <title> to this page before launch.",
   630→      evidence: [
   631→        { type: "TITLE_TEXT", label: "Page title", value: "(empty)" },
   632→        { type: "HTML_SNIPPET", label: "HTML inspected", value: `${fetch.html.length} characters` },
   633→      ],
   634→    });
   635→  }
   636→
   637→  if (content.noindex) {
   638→    findings.push({
   639→      key: "noindex",
   640→      checkId: CHECK_IDS.content,
   641→      severity: "warning",
   642→      confidence: "confirmed",
   643→      title: "Page is marked noindex",
   644→      summary: `The page tells search engines not to index it. ${spendText} of campaign spend points here.`,
   645→      explanation:
   646→        "A robots meta rule contains noindex. This does not block paid traffic, but it keeps the page out of search results and some link previews. On a campaign landing page this is sometimes intentional.",
   647→      recommendation: "Confirm the noindex rule is intentional for this campaign page.",
   648→      evidence: [{ type: "META_ROBOTS", label: "Meta robots", value: "noindex" }],
   649→    });
   650→  }
   651→
   652→  if (content.maintenance.match) {
   653→    findings.push({
   654→      key: "maintenance_wording",
   655→      checkId: CHECK_IDS.content,
   656→      severity: "warning",
   657→      confidence: "high",
   658→      title: "Page shows maintenance or unavailable wording",
   659→      summary: `Matched text: "${content.maintenance.evidence ?? content.maintenance.phrase}". ${spendText} of campaign spend points here.`,
   660→      explanation:
   661→        "The page returned HTTP 200 but its text matches an unavailable/maintenance pattern. Paid visitors would see this wording instead of the intended page.",
   662→      recommendation: "Restore the intended page content before launch, then re-scan.",
   663→      evidence: [
   664→        { type: "MATCHED_TEXT", label: "Matched text", snippet: content.maintenance.evidence ?? undefined, value: content.maintenance.phrase ?? undefined },
   665→        { type: "HTTP_STATUS", label: "HTTP status", value: String(fetch.httpStatus) },
   666→      ],
   667→    });
   668→  }
   669→
   670→  if (content.soft404.match) {
   671→    findings.push({
   672→      key: "soft_404",
   673→      checkId: CHECK_IDS.content,
   674→      severity: "warning",
   675→      confidence: "heuristic",
   676→      title: "Possible soft 404",
   677→      summary: `The server returned HTTP 200, but the page text matches not-found wording. ${spendText} of campaign spend points here.`,
   678→      explanation:
   679→        "This is a heuristic finding. The page returned HTTP 200 while its title or prominent text contains not-found language. Verify in a browser before treating it as a confirmed failure.",
   680→      recommendation: "Open the destination in a normal browser and confirm what visitors see.",
   681→      evidence: [
   682→        { type: "HTTP_STATUS", label: "HTTP status", value: String(fetch.httpStatus) },
   683→        { type: "MATCHED_TEXT", label: "Signals", value: content.soft404.signals.join("; ").slice(0, 300), snippet: content.soft404.evidence ?? undefined },
   684→      ],
   685→    });
   686→  }
   687→
   688→  if (content.soldOut.match) {
   689→    findings.push({
   690→      key: "sold_out_wording",
   691→      checkId: CHECK_IDS.content,
   692→      severity: "warning",
   693→      confidence: "high",
   694→      title: "Sold-out or unavailable product wording detected",
   695→      summary: `Matched text: "${content.soldOut.evidence ?? content.soldOut.phrase}". ${spendText} of campaign spend points here.`,
   696→      explanation:
   697→        "The page text indicates the product is unavailable. The scanner reports page wording only — it does not know your inventory state.",
   698→      recommendation: "Confirm stock levels and pause or redirect campaigns if the product is unavailable.",
   699→      evidence: [
   700→        { type: "MATCHED_TEXT", label: "Matched text", snippet: content.soldOut.evidence ?? undefined, value: content.soldOut.phrase ?? undefined },
   701→        { type: "HTTP_STATUS", label: "HTTP status", value: String(fetch.httpStatus) },
   702→      ],
   703→    });
   704→  }
   705→
   706→  if (content.bodyTextLength < 200 && !content.soft404.match) {
   707→    findings.push({
   708→      key: "thin_content",
   709→      checkId: CHECK_IDS.content,
   710→      severity: "warning",
   711→      confidence: "heuristic",
   712→      title: "Extremely thin page content",
   713→      summary: `The visible body text is ${content.bodyTextLength} characters. ${spendText} of campaign spend points here.`,
   714→      explanation:
   715→        "The retrieved HTML contains very little visible text. The page may be broken, may require JavaScript to render its content, or may genuinely be minimal. This is a heuristic finding.",
   716→      recommendation: "Verify the rendered page in a browser. If content renders via JavaScript, treat this finding as informational.",
   717→      evidence: [
   718→        { type: "HTML_SNIPPET", label: "Body text length", value: `${content.bodyTextLength} characters` },
   719→        { type: "HTML_SNIPPET", label: "HTML size inspected", value: `${fetch.bytesInspected} bytes` },
   720→      ],
   721→    });
   722→  }
   723→
   724→  if (!content.hasCta) {
   725→    findings.push({
   726→      key: "cta_absent",
   727→      checkId: CHECK_IDS.content,
   728→      severity: "info",
   729→      confidence: "heuristic",
   730→      title: "No conventional call-to-action detected",
   731→      summary: "No button or prominent link matched common CTA wording in the retrieved HTML.",
   732→      explanation:
   733→        "Some valid landing pages use unconventional wording or image-based CTAs, or require JavaScript to render. Absence of a detected CTA is not a failure by itself.",
   734→      evidence: [{ type: "CTA_CHECK", label: "CTA scan", value: "No match in buttons or prominent links" }],
   735→    });
   736→  }
   737→
   738→  if (!content.metaDescription) {
   739→    findings.push({
   740→      key: "meta_description_missing",
   741→      checkId: CHECK_IDS.content,
   742→      severity: "info",
   743→      confidence: "confirmed",
   744→      title: "Meta description is missing",
   745→      summary: "The page has no meta description.",
   746→      explanation: "Minor SEO metadata issue. Not a landing-page failure.",
   747→      evidence: [{ type: "HTML_SNIPPET", label: "Meta description", value: "(absent)" }],
   748→    });
   749→  }
   750→
   751→  return findings;
   752→}

   758→function performanceFindings(fetch: Extract<PageFetchResult, { kind: "ok" }>, ctx: ScanContext): FindingDraft[] {
   759→  const threshold = PRODUCT.scanner.slowResponseMs;
   760→  if (fetch.responseTimeMs < threshold) return [];
   761→  const spendText = fmtSpend(ctx.associatedSpendMinor, ctx.currency);
   762→  return [
   763→    {
   764→      key: "slow_response",
   765→      checkId: CHECK_IDS.performance,
   766→      severity: "warning",
   767→      confidence: "confirmed",
   768→      title: "Slow server response duration",
   769→      summary: `Server response duration: ${(fetch.responseTimeMs / 1000).toFixed(1)} s. ${spendText} of campaign spend points here.`,
   770→      explanation: `Measured from the scanner region. This is not a Core Web Vitals measurement and not a Lighthouse result. The threshold for this finding is ${(threshold / 1000).toFixed(0)} s.`,
   771→      recommendation: "Check origin performance (hosting, caching, backend) before launch.",
   772→      evidence: [
   773→        { type: "RESPONSE_TIME", label: "Server response duration", value: `${fetch.responseTimeMs} ms`, meta: { thresholdMs: threshold } },
   774→        { type: "HTTP_STATUS", label: "HTTP status", value: String(fetch.httpStatus) },
   775→      ],
   776→    },
   777→  ];
   778→}

   780→/* -------------------------------------------------------------- */
   781→/* Expectation findings (user-declared expected state)            */
   782→/* -------------------------------------------------------------- */
   783→
   784→function expectationFindings(
   785→  fetch: Extract<PageFetchResult, { kind: "ok" }>,
   786→  ctx: ScanContext,
   787→  content: ReturnType<typeof inspectContent>,
   788→  trackers: TrackerDetection[]
   789→): FindingDraft[] {
   790→  const findings: FindingDraft[] = [];
   791→  const exp = ctx.expectations;
   792→  if (!exp) return findings;
   793→  const spendText = fmtSpend(ctx.associatedSpendMinor, ctx.currency);
   794→
   795→  if (exp.expectedForm === true && !content.hasForm) {
   796→    findings.push({
   797→      key: "expected_form_missing",
   798→      checkId: CHECK_IDS.content,
   799→      severity: "warning",
   800→      confidence: "confirmed",
   801→      title: "Expected form not detected",
   802→      summary: `You declared that this destination should contain a form. No <form> element or known form embed was detected. ${spendText} of campaign spend points here.`,
   803→      explanation:
   804→        "The expected-state check compares your declaration with the retrieved HTML. JavaScript-rendered forms are not visible to this static scan.",
   805→      recommendation: "Verify the rendered page in a browser before treating this as a confirmed failure.",
   806→      evidence: [{ type: "FORM_CHECK", label: "Form scan", value: "No <form> or known form embed detected" }],
   807→    });
   808→  }
   809→
   810→  if (exp.expectedTracker) {
   811→    const t = trackers.find((tr) => tr.key === exp.expectedTracker);
   812→    if (t && !t.detected) {
   813→      findings.push({
   814→        key: `expected_tracker_${exp.expectedTracker}`,
   815→        checkId: CHECK_IDS.tracking,
   816→        severity: "warning",
   817→        confidence: "needs_verification",
   818→        title: `Expected tracker not detected: ${trackerLabel(exp.expectedTracker)}`,
   819→        summary: `You declared that ${trackerLabel(exp.expectedTracker)} should be present. It was not detected in the retrieved HTML.`,
   820→        explanation:
   821→          "The expected-state check compares your declaration with static signatures. A tag can load at runtime through a tag manager.",
   822→        recommendation: "Verify with a tag debugger on the rendered page.",
   823→        evidence: [
   824→          { type: "TRACKER_ABSENT", label: `Expected: ${trackerLabel(exp.expectedTracker)}`, value: "Not detected in retrieved HTML" },
   825→          trackingEvidenceLine(trackers, "gtm"),
   826→        ],
   827→      });
   828→    }
   829→  }
   830→
   831→  if (exp.expectedText && fetch.html.length > 0) {
   832→    const present = fetch.html.toLowerCase().includes(exp.expectedText.toLowerCase());
   833→    if (!present) {
   834→      findings.push({
   835→        key: "expected_text_missing",
   836→        checkId: CHECK_IDS.content,
   837→        severity: "warning",
   838→        confidence: "confirmed",
   839→        title: "Expected text not found",
   840→        summary: `You declared this destination should contain the text "${exp.expectedText}". It was not found in the retrieved HTML.`,
   841→        explanation: "The expected-state check searches the retrieved HTML for your declared text.",
   842→        recommendation: "Confirm the page still contains the intended copy.",
   843→        evidence: [{ type: "MATCHED_TEXT", label: "Expected text", value: exp.expectedText }],
   844→      });
   845→    }
   846→  }
   847→
   848→  if (exp.expectedHostname) {
   849→    try {
   850→      const final = new URL(fetch.finalUrl);
   851→      if (final.hostname.toLowerCase() !== exp.expectedHostname.toLowerCase()) {
   852→        findings.push({
   853→          key: "expected_hostname_mismatch",
   854→          checkId: CHECK_IDS.redirects,
   855→          severity: "warning",
   856→          confidence: "confirmed",
   857→          title: "Final hostname differs from the expected hostname",
   858→          summary: `You declared the final destination should be "${exp.expectedHostname}". The scan finished on "${final.hostname}".`,
   859→          explanation: "The expected-state check compares your declared hostname with the final URL after redirects.",
   860→          recommendation: "Check the redirect rules if the declared hostname is the intended one.",
   861→          evidence: [{ type: "FINAL_URL", label: "Final URL", value: fetch.finalUrl }],
   862→        });
   863→      }
   864→    } catch {
   865→      /* skip malformed */
   866→    }
   867→  }
   868→
   869→  return findings;
   870→}

   872→/* -------------------------------------------------------------- */
   873→/* Score + preflight (deterministic)                              */
   874→/* -------------------------------------------------------------- */
   875→
   876→export function destinationScore(findings: { severity: Severity }[]): number {
   877→  let score = 100;
   878→  for (const f of findings) {
   879→    if (f.severity === "critical") score -= 40;
   880→    else if (f.severity === "warning") score -= 12;
   881→    // Info findings do not reduce the score (documented rule).
   882→  }
   883→  return Math.max(0, score);
   884→}
   885→
   886→export function preflightStatus(findings: { severity: Severity }[]): "DO_NOT_LAUNCH" | "REVIEW_BEFORE_LAUNCH" | "LAUNCH_READY" {
   887→  if (findings.some((f) => f.severity === "critical")) return "DO_NOT_LAUNCH";
   888→  if (findings.some((f) => f.severity === "warning")) return "REVIEW_BEFORE_LAUNCH";
   889→  return "LAUNCH_READY";
   890→}
