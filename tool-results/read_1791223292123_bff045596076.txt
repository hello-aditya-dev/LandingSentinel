   527→   527→function trackingFindings(ctx: ScanContext, trackers: TrackerDetection[]): FindingDraft[] {
   528→   528→  const findings: FindingDraft[] = [];

   534→   534→    const relevant = platformRelevantTrackers(platform);
   535→   535→    if (relevant.length === 0) continue;
   536→   536→
   537→   537→    const anyDetected = relevant.some((key) => trackers.find((t) => t.key === key)?.detected);
   538→   538→    if (anyDetected) continue;
   539→   539→
   540→   540→    const platformName: Record<string, string> = {
   541→   541→      meta: "Meta",

   555→   555→    if (!gtmPresent) {

   578→   578→        evidence: baseEvidence,

   585→   585→  if (detected.length > 0) {
   586→   586→    findings.push({
   587→   587→      key: "tracking_detected",

   593→   593→      explanation:
   594→   594→        "Static detection proves the signature exists in the served HTML. It does not confirm that tags fire at runtime or that they are configured correctly.",
   595→   595→      evidence: detected.map((t) => ({
   596→   596→        type: "TRACKER_SIGNATURE",
   597→   597→        label: trackerLabel(t.key),
   598→   598→        value: `Detected — signature "${t.signatureId}"`,
   599→   599→        snippet: t.excerpt ?? undefined,
   600→   600→      })),
   601→   601→    });
   602→   602→  }
   603→   603→
   604→   604→  return findings;
   605→   605→}

   793→   793→  const spendText = fmtSpend(ctx.associatedSpendMinor, ctx.currency);
   794→   794→
   795→   795→  if (exp.expectedForm === true && !content.hasForm) {
   796→   796→    findings.push({
